struct Annotation {

};

enum class ExcerptError {
    KeinTitel,
    KeineAnmerkung,
    AnmerkungNichtEindeutig,
    ZielChatNichtEindeutig,
    OriginalNichtAktualisierbar
};

class ExcerptWorkflow {
public:
    std::expected<void, ExcerptError> execute(ExcerptRequest request) {
        if (request.titel.empty()) {
            return std::unexpected(ExcerptError::KeinTitel);
        }

        if (request.anmerkungen.empty()) {
            return std::unexpected(ExcerptError::KeineAnmerkung);
        }

        const auto zielChat = resolveTargetChat(request.chatnameZiel);
        if (!zielChat) {
            return std::unexpected(ExcerptError::ZielChatNichtEindeutig);
        }

        // Jede Anmerkung muss einen zusammenhängenden Bereich im aktuellen
        // Chatverlauf eindeutig bezeichnen.
        const auto bereiche = resolveAnnotations(request.anmerkungen);
        if (!bereiche) {
            return std::unexpected(ExcerptError::AnmerkungNichtEindeutig);
        }

        // Original.html wird ausschließlich zum Abgleich des sichtbaren
        // aktuellen Chatverlaufs aktualisiert.
        if (!updateOriginalHtml()) {
            return std::unexpected(ExcerptError::OriginalNichtAktualisierbar);
        }

        const auto quellenanker = confirmSourceAnchors(*bereiche);

        ExcerptSection auszug;
        auszug.titel = request.titel;
        auszug.kurzbeschreibung = describe(*bereiche);
        auszug.inhalt = preserveOriginalMarkdown(*bereiche);
        auszug.quellenLinks = createRelativeLinks(zielChat->excerptFile,
                                                   quellenanker);

        appendToExcerptFile(zielChat->excerptFile, auszug);
        updateExcerptTableOfContents(zielChat->excerptFile);

        return {};
    }

private:
    struct ChatTarget {
        std::string excerptFile;
    };

    struct ChatRange {};
    struct SourceAnchor {};
    struct ExcerptSection {
        std::string titel;
        std::string kurzbeschreibung;
        std::string inhalt;
        std::vector<std::string> quellenLinks;
    };

    std::optional<ChatTarget> resolveTargetChat(
        const std::optional<std::string>& chatnameZiel);

    std::optional<std::vector<ChatRange>> resolveAnnotations(
        const std::vector<Annotation>& anmerkungen);

    bool updateOriginalHtml();
    std::vector<SourceAnchor> confirmSourceAnchors(
        const std::vector<ChatRange>& bereiche);

    std::string preserveOriginalMarkdown(
        const std::vector<ChatRange>& bereiche);

    std::string describe(const std::vector<ChatRange>& bereiche);

    std::vector<std::string> createRelativeLinks(
        const std::string& excerptFile,

                             const ExcerptSection& auszug);

    void updateExcerptTableOfContents(const std::string& excerptFile);
};
